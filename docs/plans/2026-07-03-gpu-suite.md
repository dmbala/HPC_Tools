# GPU Suite Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add three Prometheus-backed window/fleet GPU reporting tools — `xid_history`, `energy_report`, `fleet_util` — to `gpu/`, per the approved spec (throttle_report is out of scope: its DCGM field is not scraped on this cluster).

**Architecture:** Each tool is an extensionless Python executable with a pure aggregation core (offline-testable) and a `--from-json` replay seam; a shared `gpu/gpulib.py` holds the jobstats-config import, instant/range Prometheus query helpers, the UUID-casing join, window parsing, and `err_exit`. Slurm enrichment (sacct/sinfo) is read via subprocess with timeouts.

**Tech Stack:** Python 3.6-compatible + `requests` (ships with jobstats); stdlib `unittest`; Prometheus HTTP API; `sacct`/`sinfo`.

**Spec:** `docs/specs/2026-07-03-gpu-suite-design.md` (approved).

**Branch:** create `gpu-suite` FROM the `gpu-health` branch (it has `gpu/tests/` scaffolding, 50 passing tests, and the repo `.gitignore`). The controller does this before Task 1.

## Global Constraints

- Python 3.6-compatible; stdlib + `requests` only. No f-strings, walrus, dataclasses, or `subprocess.run(capture_output=/text=)` kwargs — use `stdout=subprocess.PIPE, stderr=subprocess.PIPE, universal_newlines=True`.
- Tools are `gpu/xid_history`, `gpu/energy_report`, `gpu/fleet_util` — no `.py` extension, `#!/usr/bin/env python3`, executable bit set.
- `gpu/gpulib.py` must import cleanly WITHOUT the jobstats config or `requests` present (both are imported lazily inside functions) so tests run anywhere.
- Exit codes: `0` clean/report, `1` findings (`xid_history` only), `3` probe/setup error via `gpulib.err_exit` (`<tool>: error: <msg>` on stderr). argparse usage errors stay exit 2.
- Error messages must NEVER contain the Prometheus endpoint URL (it embeds credentials).
- Every subprocess/HTTP call has a timeout.
- Truncation is never silent (`--max-jobs` reports the exact skipped count); unmatched data is counted and reported, never guessed.
- UUID join rule: `labels.get("UUID") or labels.get("uuid")` (DCGM uppercase, jobstats exporter lowercase).
- Tests: `python3 -m unittest discover -s gpu/tests -v` from the repo root. Base (gpu-health branch): 50 tests.
- Commit messages: short imperative subject, no attribution footers of any kind.

```python
"""Load the extensionless gpu tools as modules; locate fixtures."""
import importlib.machinery
import importlib.util
import os

HERE = os.path.dirname(os.path.abspath(__file__))
FIXTURES = os.path.join(HERE, "fixtures")


def load_tool(name):
    path = os.path.abspath(os.path.join(HERE, "..", name))
    loader = importlib.machinery.SourceFileLoader(name, path)
    spec = importlib.util.spec_from_loader(name, loader)
    mod = importlib.util.module_from_spec(spec)
    loader.exec_module(mod)
    return mod


def fixture(name):
    with open(os.path.join(FIXTURES, name)) as fh:
        return fh.read()
```

```python
"""Shared helpers for the gpu/ Prometheus window/fleet tools.

Imports of the jobstats `config` module and `requests` are lazy (inside
functions) so this module -- and the tools' pure cores -- import cleanly on
machines without a jobstats install (e.g. in the offline test suite).
"""
import datetime
import json
import sys
import time

for _p in ("/usr/local/bin", "/usr/bin"):
    if _p not in sys.path:
        sys.path.append(_p)


def err_exit(tool, msg):
    sys.stderr.write("%s: error: %s\n" % (tool, msg))
    sys.exit(3)


def load_prom_server(tool):
    """PROM_SERVER from the jobstats config; err_exit if unavailable."""
    try:
        from config import PROM_SERVER
        return PROM_SERVER
    except Exception as exc:
        err_exit(tool, "cannot import jobstats config (PROM_SERVER): %s -- "
                 "run where the jobstats command lives" % exc)


def _get(prom, path, params, timeout, kind):
    """Shared HTTP plumbing. Never lets the endpoint URL into an error."""
    import requests
    try:
        resp = requests.get(prom + path, params=params, timeout=timeout)
    except requests.RequestException as exc:
        raise RuntimeError("prometheus %s failed: %s"
                           % (kind, exc.__class__.__name__))
    if resp.status_code != 200:
        raise RuntimeError("prometheus %s failed: HTTP %d"
                           % (kind, resp.status_code))
    payload = resp.json()
    if payload.get("status") != "success":
        raise RuntimeError("prometheus %s failed: %s"
                           % (kind, payload.get("error", "non-success")))
    return payload["data"]["result"]


def query(prom, promql, when=None, timeout=60):
    """Instant query -> result list. Raises RuntimeError on failure."""
    params = {"query": promql}
    if when is not None:
        params["time"] = when
    return _get(prom, "/api/v1/query", params, timeout, "query")


def query_range(prom, promql, start, end, step, timeout=120):
    """Range query -> matrix list. Raises RuntimeError on failure."""
    return _get(prom, "/api/v1/query_range",
                {"query": promql, "start": start, "end": end, "step": step},
                timeout, "range query")


def series_uuid(labels):
    """UUID from series labels, either casing (DCGM: UUID, exporter: uuid)."""
    return labels.get("UUID") or labels.get("uuid")


def iso_epoch(text):
    """'YYYY-mm-ddTHH:MM:SS' -> local-time epoch int. Raises ValueError."""
    return int(time.mktime(datetime.datetime.strptime(
        text, "%Y-%m-%dT%H:%M:%S").timetuple()))


def add_window_args(parser):
    parser.add_argument("--days", type=int, default=7,
                        help="window length ending now (default 7)")
    parser.add_argument("--since", metavar="ISO",
                        help="window start, YYYY-mm-ddTHH:MM:SS")
    parser.add_argument("--until", metavar="ISO",
                        help="window end, YYYY-mm-ddTHH:MM:SS")


def parse_window(args, tool):
    """(--days | --since+--until) -> (start_epoch, end_epoch); exit 3 on bad."""
    if args.since or args.until:
        if not (args.since and args.until):
            err_exit(tool, "--since and --until must be given together")
        try:
            start, end = iso_epoch(args.since), iso_epoch(args.until)
        except ValueError:
            err_exit(tool, "timestamps must be YYYY-mm-ddTHH:MM:SS")
        if end <= start:
            err_exit(tool, "--until must be after --since")
        return start, end
    end = int(time.time())
    return end - args.days * 86400, end


def load_bundle(path, tool):
    """Read a --from-json replay bundle; exit 3 if unreadable."""
    try:
        with open(path) as fh:
            return json.load(fh)
    except (OSError, ValueError) as exc:
        err_exit(tool, "cannot read replay bundle: %s" % exc)
```

```python
#!/usr/bin/env python3
"""xid_history -- window report of GPU XID errors fleet-wide, each mapped to
the node, GPU, and the Slurm job/user holding the GPU at the time.

Reads DCGM_FI_DEV_XID_ERRORS from the same Prometheus the jobstats stack
feeds (endpoint from the jobstats config module), joins to jobs via the
nvidia_gpu_jobId series (UUID match), and enriches with one batched sacct
call. Runs wherever jobstats is installed -- no allocation needed.

Usage:
    xid_history [--days N | --since ISO --until ISO] [--node NODE] [--json]

Exit codes: 0 no XIDs in the window, 1 XIDs found, 3 probe/setup error.
"""

import argparse
import datetime
import json
import os
import subprocess
import sys

sys.path.insert(0, os.path.dirname(os.path.realpath(__file__)))
import gpulib

TOOL = "xid_history"
STEP_S = 300      # range-query resolution
GAP_STEPS = 2     # samples further apart than this many steps split episodes


def episodes(matrix, step_s=STEP_S, gap_steps=GAP_STEPS):
    """Compress a query_range matrix of nonzero XID samples into episodes,
    one per burst per (node, gpu, err_code), sorted by first_seen."""
    out = []
    for series in matrix:
        labels = series.get("metric", {})
        node = labels.get("Hostname") or labels.get("host") or "?"
        template = {"node": node, "gpu": labels.get("gpu", "?"),
                    "uuid": gpulib.series_uuid(labels),
                    "err_code": labels.get("err_code", "?"),
                    "err_msg": labels.get("err_msg", "")}
        current = None
        for ts, _val in series.get("values", []):
            ts = float(ts)
            if current and ts - current["last_seen"] <= gap_steps * step_s:
                current["last_seen"] = ts
            else:
                current = dict(template, first_seen=ts, last_seen=ts,
                               jobid=None, user=None, state=None)
                out.append(current)
    out.sort(key=lambda e: e["first_seen"])
    return out


def attach_jobids(eps, jobid_by_ts):
    """Fill each episode's jobid from {str(int(first_seen)): instant result}.
    Job id 0 (unallocated GPU) stays None."""
    for ep in eps:
        for series in jobid_by_ts.get(str(int(ep["first_seen"])), []):
            if gpulib.series_uuid(series.get("metric", {})) != ep["uuid"]:
                continue
            try:
                jid = int(float(series["value"][1]))
            except (TypeError, ValueError, KeyError, IndexError):
                continue
            if jid > 0:
                ep["jobid"] = jid
            break
    return eps


def attach_sacct(eps, sacct_text):
    """Fill user/state from 'JobIDRaw|User|Account|State' -P rows."""
    info = {}
    for line in (sacct_text or "").splitlines():
        parts = line.strip().split("|")
        if len(parts) >= 4 and parts[0].isdigit():
            info[int(parts[0])] = (parts[1], parts[3])
    for ep in eps:
        if ep["jobid"] in info:
            ep["user"], ep["state"] = info[ep["jobid"]]
    return eps


def _fmt(ts):
    return datetime.datetime.fromtimestamp(ts).strftime("%Y-%m-%d %H:%M")


def render(eps, start, end):
    lines = ["xid_history: %s -> %s" % (_fmt(start), _fmt(end)), ""]
    if not eps:
        lines.append("no XID errors in the window")
        return "\n".join(lines)
    row = "%-17s %-17s %-16s %-4s %-5s %-10s %-10s %-10s %s"
    lines.append(row % ("FIRST_SEEN", "LAST_SEEN", "NODE", "GPU", "XID",
                        "JOB", "USER", "STATE", "MESSAGE"))
    for ep in eps:
        lines.append(row % (_fmt(ep["first_seen"]), _fmt(ep["last_seen"]),
                            ep["node"].split(".")[0], ep["gpu"],
                            ep["err_code"], ep["jobid"] or "-",
                            ep["user"] or "-", ep["state"] or "-",
                            ep["err_msg"]))
    lines.append("")
    lines.append("%d episode(s) on %d node(s)"
                 % (len(eps), len(set(e["node"] for e in eps))))
    return "\n".join(lines)


def main(argv=None):
    parser = argparse.ArgumentParser(
        prog=TOOL,
        description="window report of GPU XID errors mapped to jobs/users "
                    "(exit 0 none, 1 found, 3 error)")
    gpulib.add_window_args(parser)
    parser.add_argument("--node", help="restrict to one node (Hostname)")
    parser.add_argument("--json", action="store_true")
    parser.add_argument("--from-json", metavar="FILE",
                        help="replay a captured query bundle (testing)")
    args = parser.parse_args(argv)
    start, end = gpulib.parse_window(args, TOOL)

    if args.from_json:
        bundle = gpulib.load_bundle(args.from_json, TOOL)
        eps = episodes(bundle.get("xid", []))
        attach_jobids(eps, bundle.get("jobid", {}))
        attach_sacct(eps, bundle.get("sacct", ""))
    else:
        prom = gpulib.load_prom_server(TOOL)
        selector = '{Hostname="%s"}' % args.node if args.node else ""
        try:
```

```python
#!/usr/bin/env python3
"""energy_report -- per-user/account GPU energy (kWh) over a window.

Lists the target's GPU jobs via sacct, then reads each job's energy from
DCGM_FI_DEV_TOTAL_ENERGY_CONSUMPTION (a monotonic mJ counter) as
max_over_time - min_over_time across the job's window, joined to the job's
GPUs the same way jobstats_dcgm does (nvidia_gpu_jobId -> UUID). Runs
wherever jobstats is installed -- no allocation needed. Report-only.

Usage:
    energy_report (-u USER | -A ACCOUNT)
                  [--days N | --since ISO --until ISO]
                  [--top N] [--max-jobs M] [--json]

Exit codes: 0 report, 3 probe/setup error.
"""

import argparse
import datetime
import json
import os
import subprocess
import sys
import time

sys.path.insert(0, os.path.dirname(os.path.realpath(__file__)))
import gpulib

TOOL = "energy_report"
MJ_PER_KWH = 3.6e9


def _elapsed_seconds(text):
    """sacct Elapsed '[DD-]HH:MM:SS' -> seconds (0 on parse failure)."""
    if not text:
        return 0
    days = 0
    if "-" in text:
        day_part, text = text.split("-", 1)
        try:
            days = int(day_part)
        except ValueError:
            return 0
    parts = text.split(":")
    try:
        hours, minutes, seconds = (int(p) for p in parts[-3:])
    except ValueError:
        return 0
    return days * 86400 + hours * 3600 + minutes * 60 + seconds


def _gpus_from_tres(tres):
    """GPU count from AllocTRES ('gres/gpu=4', or typed 'gres/gpu:h100=4')."""
    fallback = 0
    for item in (tres or "").split(","):
        if item.startswith("gres/gpu="):
            try:
                return int(item.split("=", 1)[1])
            except ValueError:
                return 0
        if item.startswith("gres/gpu:") and "=" in item:
            try:
                fallback = int(item.split("=", 1)[1])
            except ValueError:
                pass
    return fallback


def parse_sacct_jobs(text):
    """-P rows 'JobIDRaw|User|Account|Start|End|Elapsed|AllocTRES|State'
    -> GPU-job dicts (non-GPU jobs dropped)."""
    jobs = []
    for line in (text or "").splitlines():
        parts = line.strip().split("|")
        if len(parts) < 8 or not parts[0].isdigit():
            continue
        gpus = _gpus_from_tres(parts[6])
        if not gpus:
            continue
        jobs.append({"jobid": int(parts[0]), "user": parts[1],
                     "account": parts[2], "start": parts[3], "end": parts[4],
                     "elapsed_s": _elapsed_seconds(parts[5]), "gpus": gpus,
                     "state": parts[7].split()[0] if parts[7] else ""})
    return jobs


def energy_kwh(delta_result, uuids):
    """Sum a per-UUID energy-delta instant result (mJ) over the job's GPUs
    -> kWh; None when no series matched (reported as 'no data')."""
    wanted = set(uuids)
    total_mj, matched = 0.0, False
    for series in delta_result:
        if gpulib.series_uuid(series.get("metric", {})) not in wanted:
            continue
        try:
            total_mj += float(series["value"][1])
            matched = True
        except (TypeError, ValueError, KeyError, IndexError):
            continue
    return (total_mj / MJ_PER_KWH) if matched else None


def rollup(job_rows, top_n):
    counted = [j for j in job_rows if j.get("kwh") is not None]
    total_kwh = sum(j["kwh"] for j in counted)
    total_gpu_h = sum(j["elapsed_s"] * j["gpus"] for j in counted) / 3600.0
    per_user = {}
    for j in counted:
        per_user[j["user"]] = per_user.get(j["user"], 0.0) + j["kwh"]
    return {"counted": len(counted),
            "no_data": len(job_rows) - len(counted),
            "total_kwh": total_kwh,
            "total_gpu_hours": total_gpu_h,
            "mean_w_per_gpu": (total_kwh / total_gpu_h * 1000.0)
                              if total_gpu_h else 0.0,
            "top_jobs": sorted(counted, key=lambda j: j["kwh"],
                               reverse=True)[:top_n],
            "per_user_kwh": per_user}


def _end_epoch(job):
    try:
        return gpulib.iso_epoch(job["end"])
    except (ValueError, TypeError):
        return int(time.time())  # still running / Unknown


def _fmt(ts):
    return datetime.datetime.fromtimestamp(ts).strftime("%Y-%m-%d %H:%M")


def render(summary, start, end):
    lines = ["energy_report: %s -> %s" % (_fmt(start), _fmt(end))]
    lines.append("  jobs counted: %d   no data: %d   skipped by "
                 "--max-jobs: %d" % (summary["counted"], summary["no_data"],
                                     summary.get("skipped", 0)))
    lines.append("  total energy : %.2f kWh over %.1f GPU-hours "
                 "(mean %.0f W per GPU)"
                 % (summary["total_kwh"], summary["total_gpu_hours"],
                    summary["mean_w_per_gpu"]))
    if summary["top_jobs"]:
        lines.append("")
        row = "%-12s %-10s %10s %8s %8s %s"
        lines.append(row % ("JOBID", "USER", "KWH", "GPU_H", "MEAN_W",
                            "STATE"))
        for j in summary["top_jobs"]:
            gpu_h = j["elapsed_s"] * j["gpus"] / 3600.0
            mean_w = (j["kwh"] / gpu_h * 1000.0) if gpu_h else 0.0
            lines.append(row % (j["jobid"], j["user"], "%.2f" % j["kwh"],
                                "%.1f" % gpu_h, "%.0f" % mean_w, j["state"]))
    if len(summary["per_user_kwh"]) > 1:
        lines.append("")
        lines.append("per-user kWh:")
        for user, kwh in sorted(summary["per_user_kwh"].items(),
                                key=lambda kv: kv[1], reverse=True):
            lines.append("  %-12s %10.2f" % (user, kwh))
    return "\n".join(lines)


def main(argv=None):
    parser = argparse.ArgumentParser(
        prog=TOOL,
        description="per-user/account GPU energy (kWh) over a window from "
                    "DCGM counters (exit 0 report, 3 error)")
    gpulib.add_window_args(parser)
    parser.add_argument("-u", "--user")
    parser.add_argument("-A", "--account")
    parser.add_argument("--top", type=int, default=10)
    parser.add_argument("--max-jobs", type=int, default=200, dest="max_jobs",
                        help="cap on per-job Prometheus queries "
                             "(default 200; truncation is reported)")
    parser.add_argument("--json", action="store_true")
    parser.add_argument("--from-json", metavar="FILE",
                        help="replay a captured query bundle (testing)")
    args = parser.parse_args(argv)
    if bool(args.user) == bool(args.account):
        gpulib.err_exit(TOOL, "exactly one of -u USER or -A ACCOUNT required")
    start, end = gpulib.parse_window(args, TOOL)

    discovery = energy = None
    if args.from_json:
        bundle = gpulib.load_bundle(args.from_json, TOOL)
        sacct_text = bundle.get("sacct", "")
        discovery = bundle.get("discovery", {})
        energy = bundle.get("energy", {})
        prom = None
    else:
        prom = gpulib.load_prom_server(TOOL)
        cmd = ["sacct", "-X", "-n", "-P", "-o",
               "JobIDRaw,User,Account,Start,End,Elapsed,AllocTRES,State",
               "-S", datetime.datetime.fromtimestamp(start).strftime(
                   "%Y-%m-%dT%H:%M:%S"),
               "-E", datetime.datetime.fromtimestamp(end).strftime(
                   "%Y-%m-%dT%H:%M:%S")]
        if args.user:
            cmd += ["-u", args.user]
        else:
            cmd += ["-A", args.account, "-a"]
        try:
            r = subprocess.run(cmd, stdout=subprocess.PIPE,
                               stderr=subprocess.PIPE,
                               universal_newlines=True, timeout=120)
        except (OSError, subprocess.TimeoutExpired) as exc:
            gpulib.err_exit(TOOL, "sacct failed: %s" % exc)
        if r.returncode != 0:
            gpulib.err_exit(TOOL, "sacct failed: %s" % r.stderr.strip())
        sacct_text = r.stdout

    jobs = parse_sacct_jobs(sacct_text)
    skipped = max(0, len(jobs) - args.max_jobs)
    jobs = jobs[:args.max_jobs]

    for job in jobs:
        jid = str(job["jobid"])
        window_s = max(job["elapsed_s"], 60)
        if args.from_json:
            disc = discovery.get(jid, [])
        else:
            try:
                disc = gpulib.query(
                    prom, "max_over_time((nvidia_gpu_jobId == %d)[%ds:])"
                    % (job["jobid"], window_s), when=_end_epoch(job))
            except RuntimeError as exc:
                gpulib.err_exit(TOOL, str(exc))
        uuids = [u for u in (gpulib.series_uuid(s.get("metric", {}))
                             for s in disc) if u]
        if not uuids:
            job["kwh"] = None
            continue
        if args.from_json:
            eres = energy.get(jid, [])
        else:
            rx = "|".join(uuids)
            promql = ("(max_over_time(DCGM_FI_DEV_TOTAL_ENERGY_CONSUMPTION"
                      "{UUID=~\"%s\"}[%ds:]) - min_over_time("
                      "DCGM_FI_DEV_TOTAL_ENERGY_CONSUMPTION{UUID=~\"%s\"}"
                      "[%ds:]))" % (rx, window_s, rx, window_s))
            try:
                eres = gpulib.query(prom, promql, when=_end_epoch(job))
            except RuntimeError as exc:
                gpulib.err_exit(TOOL, str(exc))
        job["kwh"] = energy_kwh(eres, uuids)

    summary = rollup(jobs, args.top)
    summary["skipped"] = skipped
    if args.json:
        print(json.dumps({"start": start, "end": end, "summary": summary},
                         indent=2))
    else:
        print(render(summary, start, end))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
"""fleet_util -- what fraction of ALLOCATED GPU-hours actually did work.

Over a window, per Slurm partition: GPU-hours allocated to jobs
(nvidia_gpu_jobId > 0) vs GPU-hours whose SM activity
(DCGM_FI_PROF_SM_ACTIVE) was below a threshold. Capacity-planning evidence:
"the GPUs are all taken" vs "the GPUs are all working" are different claims.
Runs wherever jobstats is installed -- no allocation needed. Report-only.

Usage:
    fleet_util [--days N | --since ISO --until ISO] [--partition P]
               [--threshold PCT] [--step HOURS] [--json]

Exit codes: 0 report, 3 probe/setup error.
"""

import argparse
import datetime
import json
import os
import subprocess
import sys

sys.path.insert(0, os.path.dirname(os.path.realpath(__file__)))
import gpulib

TOOL = "fleet_util"
WORST_N = 5


def parse_sinfo_map(text):
    """sinfo -N -h -o '%n %P' rows -> {node: [partitions]} ('*' stripped)."""
    mapping = {}
    for line in (text or "").splitlines():
        parts = line.split()
        if len(parts) >= 2:
            mapping.setdefault(parts[0], []).append(parts[1].rstrip("*"))
    return mapping


def utilization(alloc_matrix, smact_matrix, node_to_parts, threshold_pct,
                step_s, allowed_nodes=None):
    """Join allocation and SM-activity matrices per (uuid, timestep).

    A sample is wasted when allocated AND sm*100 < threshold_pct; unmatched
    when allocated but no SM sample exists for that (uuid, ts) -- counted
    separately, never guessed. A node in several partitions is attributed
    to each; 'overall' counts each GPU-hour once.
    """
    smact = {}
    for series in smact_matrix:
        uuid = gpulib.series_uuid(series.get("metric", {}))
        for ts, val in series.get("values", []):
            try:
                smact[(uuid, float(ts))] = float(val)
            except (TypeError, ValueError):
                continue
    hours = step_s / 3600.0
    partitions = {}
    overall = {"allocated_h": 0.0, "wasted_h": 0.0, "unmatched_h": 0.0}
    node_wasted = {}
    for series in alloc_matrix:
        labels = series.get("metric", {})
        uuid = gpulib.series_uuid(labels)
        node = (labels.get("host") or labels.get("Hostname")
                or "?").split(":")[0].split(".")[0]
        if allowed_nodes is not None and node not in allowed_nodes:
            continue
        parts = node_to_parts.get(node, ["(unknown)"])
        for ts, val in series.get("values", []):
            try:
                if float(val) <= 0:
                    continue
            except (TypeError, ValueError):
                continue
            sm = smact.get((uuid, float(ts)))
            wasted = sm is not None and sm * 100.0 < threshold_pct
            unmatched = sm is None
            overall["allocated_h"] += hours
            if wasted:
                overall["wasted_h"] += hours
                node_wasted[node] = node_wasted.get(node, 0.0) + hours
            if unmatched:
                overall["unmatched_h"] += hours
            for part in parts:
                stats = partitions.setdefault(
                    part, {"allocated_h": 0.0, "wasted_h": 0.0,
                           "unmatched_h": 0.0})
                stats["allocated_h"] += hours
                if wasted:
                    stats["wasted_h"] += hours
                if unmatched:
                    stats["unmatched_h"] += hours
    worst = sorted(node_wasted.items(), key=lambda kv: kv[1],
                   reverse=True)[:WORST_N]
    return {"partitions": partitions, "overall": overall,
            "worst_nodes": worst, "threshold_pct": threshold_pct}


def _fmt(ts):
    return datetime.datetime.fromtimestamp(ts).strftime("%Y-%m-%d %H:%M")


def render(result, start, end, step_h):
    lines = ["fleet_util: %s -> %s  (step %gh, wasted = SM_ACT < %g%%)"
             % (_fmt(start), _fmt(end), step_h, result["threshold_pct"]), ""]
    row = "%-24s %12s %12s %8s %12s"
    lines.append(row % ("PARTITION", "ALLOC_GPU_H", "WASTED_GPU_H",
                        "WASTED%", "UNMATCHED_H"))

    def emit(name, stats):
        pct = (100.0 * stats["wasted_h"] / stats["allocated_h"]) \
            if stats["allocated_h"] else 0.0
        lines.append(row % (name, "%.1f" % stats["allocated_h"],
                            "%.1f" % stats["wasted_h"], "%.1f" % pct,
                            "%.1f" % stats["unmatched_h"]))

    for part in sorted(result["partitions"]):
        emit(part, result["partitions"][part])
    emit("(overall)", result["overall"])
    if result["worst_nodes"]:
        lines.append("")
        lines.append("worst nodes by wasted GPU-hours:")
        for node, wasted_h in result["worst_nodes"]:
            lines.append("  %-20s %10.1f" % (node, wasted_h))
    lines.append("")
    lines.append("note: a node in several partitions is counted in each; "
                 "(overall) counts each GPU-hour once")
    return "\n".join(lines)


def main(argv=None):
    parser = argparse.ArgumentParser(
        prog=TOOL,
        description="fraction of allocated GPU-hours with SM activity below "
                    "a threshold, by partition (exit 0 report, 3 error)")
    gpulib.add_window_args(parser)
    parser.add_argument("--partition", help="restrict to one partition")
    parser.add_argument("--threshold", type=float, default=10.0,
                        help="wasted when SM_ACT%% is below this (default 10)")
    parser.add_argument("--step", type=float, default=1.0,
                        help="sample step in hours (default 1)")
    parser.add_argument("--json", action="store_true")
    parser.add_argument("--from-json", metavar="FILE",
                        help="replay a captured query bundle (testing)")
    args = parser.parse_args(argv)
    start, end = gpulib.parse_window(args, TOOL)
    step_s = int(args.step * 3600)

    if args.from_json:
        bundle = gpulib.load_bundle(args.from_json, TOOL)
        alloc = bundle.get("alloc", [])
        smact = bundle.get("smact", [])
        sinfo_text = bundle.get("sinfo", "")
    else:
        prom = gpulib.load_prom_server(TOOL)
        try:
            alloc = gpulib.query_range(prom, "nvidia_gpu_jobId > 0",
                                       start, end, step_s)
            smact = gpulib.query_range(prom, "DCGM_FI_PROF_SM_ACTIVE",
                                       start, end, step_s)
        except RuntimeError as exc:
            gpulib.err_exit(TOOL, str(exc))
        try:
            r = subprocess.run(["sinfo", "-N", "-h", "-o", "%n %P"],
                               stdout=subprocess.PIPE,
                               stderr=subprocess.PIPE,
                               universal_newlines=True, timeout=60)
        except (OSError, subprocess.TimeoutExpired) as exc:
            gpulib.err_exit(TOOL, "sinfo failed: %s" % exc)
        if r.returncode != 0:
            gpulib.err_exit(TOOL, "sinfo failed: %s" % r.stderr.strip())
        sinfo_text = r.stdout

    node_to_parts = parse_sinfo_map(sinfo_text)
    allowed = None
    if args.partition:
        allowed = set(node for node, parts in node_to_parts.items()
                      if args.partition in parts)
        if not allowed:
            gpulib.err_exit(TOOL, "no nodes found in partition %s"
                            % args.partition)
    result = utilization(alloc, smact, node_to_parts, args.threshold,
                         step_s, allowed_nodes=allowed)
    if args.json:
        print(json.dumps({"start": start, "end": end, "result": result},
                         indent=2))
    else:
        print(render(result, start, end, args.step))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
